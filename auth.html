const auth = firebase.auth(); const realtimeDB = firebase.database(); // SIGN UP document.getElementById("signupBtn").onclick = async () => { const email = document.getElementById("email").value; const password = document.getElementById("password").value; try { await auth.createUserWithEmailAndPassword(email, password); document.getElementById("usernameSection").style.display = "block"; } catch (err) { alert("Signup error: " + err.message); } }; // SAVE USERNAME document.getElementById("saveUsernameBtn").onclick = async () => { const username = document.getElementById("username").value.trim(); const uid = auth.currentUser.uid; if (!username) { alert("Username cannot be empty."); return; } // Check if username is taken const allUsers = await realtimeDB.ref("users").get(); let taken = false; allUsers.forEach(child => { if (child.val().username === username) { taken = true; } }); if (taken) { alert("Username already taken."); return; } // Save user await realtimeDB.ref("users/" + uid).set({ uid: uid, email: auth.currentUser.email, username: username }); localStorage.setItem("username", username); localStorage.setItem("email", auth.currentUser.email); window.location.href = "index.html"; }; // LOGIN document.getElementById("loginBtn").onclick = async () => { const email = document.getElementById("email").value; const password = document.getElementById("password").value; try { const userCred = await auth.signInWithEmailAndPassword(email, password); const uid = userCred.user.uid; const snap = await realtimeDB.ref("users/" + uid).get(); if (!snap.exists()) { document.getElementById("usernameSection").style.display = "block"; } else { const data = snap.val(); localStorage.setItem("username", data.username); localStorage.setItem("email", data.email); window.location.href = "index.html"; } } catch (err) { alert("Login error: " + err.message); } };
